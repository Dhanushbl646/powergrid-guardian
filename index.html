Install Node.js

2

Install dependencies

cd backend && npm install

3

Start the server

Open the frontend

5

(Optional) Twilio SMS

📄 server.js — Full Backend Code Node.js + Express
// backend/server.js const express = require('express'); const cors = require('cors'); const fs = require('fs'); const multer = require('multer'); // const twilio = require('twilio'); // Uncomment for real SMS const app = express(); const PORT = 3000; const upload = multer({ dest: 'uploads/' }); app.use(cors()); app.use(express.json()); // ── Simple JSON "database" helpers ── const DB_PATH = './data/linemen.json'; const readDB = () => JSON.parse(fs.readFileSync(DB_PATH, 'utf-8')); const writeDB = (data) => fs.writeFileSync(DB_PATH, JSON.stringify(data, null, 2)); app.get('/api/linemen', (req, res) => { res.json(readDB()); }); app.post('/api/linemen', (req, res) => { const { name, phone, email, location, tcs } = req.body; const db = readDB(); const newLineman = { id: `lm_${Date.now()}`, name, phone, email, location, tcs, createdAt: new Date() }; db.push(newLineman); writeDB(db); res.status(201).json(newLineman); }); app.get('/api/linemen/tc/:tcId', (req, res) => { const db = readDB(); const lineman = db.find(lm => lm.tcs.includes(req.params.tcId.toUpperCase())); if (!lineman) return res.status(404).json({ error: 'No line man found' }); res.json(lineman); }); app.post('/api/reports', upload.single('photo'), (req, res) => { const { tcId, location, description } = req.body; const db = readDB(); const lineman = db.find(lm => lm.tcs.includes(tcId?.toUpperCase())); console.log(`SMS sent to ${lineman?.name}: Issue at ${location} - ${description}`); res.json({ success: true, assignedTo: lineman || null }); }); app.post('/api/sensors/alert', (req, res) => { const { tcId, problem, severity, location } = req.body; const mapsLink = `https://maps.google.com/?q=${encodeURIComponent(location)}`; const smsMsg = `ALERT: ${problem} at ${tcId}. Maps: ${mapsLink}`; console.log(`[SMS] ${smsMsg}`); res.json({ success: true, smsMessage: smsMsg, mapsLink }); }); app.listen(PORT, () => console.log(`✅ Server on http://localhost:${PORT}`));
